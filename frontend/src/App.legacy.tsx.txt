const API_BASE_CANDIDATES = Array.from(
  new Set([
    "http://127.0.0.1:8001",
    "http://127.0.0.1:8000",
  ].filter(Boolean) as string[]),
);
let activeApiBase = API_BASE_CANDIDATES[0];
let accessToken: string | null = null;
let refreshPromise: Promise<AuthTokenResponse | null> | null = null;
let authenticationFailed = () => {};

function cookieValue(name: string) {
  const prefix = `${encodeURIComponent(name)}=`;
  const item = document.cookie.split("; ").find((cookie) => cookie.startsWith(prefix));
  return item ? decodeURIComponent(item.slice(prefix.length)) : null;
}

async function refreshAccessToken(): Promise<AuthTokenResponse | null> {
  if (refreshPromise) {
    return refreshPromise;
  }
  refreshPromise = (async () => {
    const csrfToken = cookieValue("icu_csrf");
    if (!csrfToken) {
      return null;
    }
    for (const base of [activeApiBase, ...API_BASE_CANDIDATES.filter((item) => item !== activeApiBase)]) {
      try {
        const response = await fetch(`${base}/api/auth/refresh`, {
          method: "POST",
          credentials: "include",
          headers: { "X-CSRF-Token": csrfToken },
        });
        if (!response.ok) {
          continue;
        }
        const payload: AuthTokenResponse = await response.json();
        accessToken = payload.access_token;
        activeApiBase = base;
        return payload;
      } catch {
        // Try the next configured local backend.
      }
    }
    accessToken = null;
    return null;
  })().finally(() => {
    refreshPromise = null;
  });
  return refreshPromise;
}

async function apiFetch(path: string, init?: RequestInit, allowRefresh = true): Promise<Response> {
  let lastError: unknown;
  const orderedBases = [activeApiBase, ...API_BASE_CANDIDATES.filter((base) => base !== activeApiBase)];
  for (const base of orderedBases) {
    try {
      const headers = new Headers(init?.headers);
      if (accessToken) {
        headers.set("Authorization", `Bearer ${accessToken}`);
      }
      const response = await fetch(`${base}${path}`, {
        ...init,
        credentials: "include",
        headers,
      });
      activeApiBase = base;
      const refreshEligible = path !== "/api/auth/login" && path !== "/api/auth/refresh";
      if (response.status === 401 && allowRefresh && refreshEligible) {
        const refreshed = await refreshAccessToken();
        if (refreshed) {
          return apiFetch(path, init, false);
        }
        authenticationFailed();
      }
      return response;
    } catch (err) {
      lastError = err;
    }
  }
  throw lastError;
}

function stableColor(value: string, palette: string[]) {
  const numeric = value.split("").reduce((total, character) => total + character.charCodeAt(0), 0);
  return palette[numeric % palette.length];
}

function hospitalCode(name: string) {
  const words = name.replace(/Sri Lanka|Hospital|Teaching|General|Central|for Women/gi, "").trim().split(/\s+/);
  return (
    words
      .slice(0, 3)
      .map((word) => word[0])
      .join("")
      .toUpperCase() || "H"
  );
}

function bearingDirection(from: [number, number], to: [number, number]) {
  const y = Math.sin((to[1] - from[1]) * Math.PI / 180) * Math.cos(to[0] * Math.PI / 180);
  const x =
    Math.cos(from[0] * Math.PI / 180) * Math.sin(to[0] * Math.PI / 180)
    - Math.sin(from[0] * Math.PI / 180)
      * Math.cos(to[0] * Math.PI / 180)
      * Math.cos((to[1] - from[1]) * Math.PI / 180);
  const bearing = (Math.atan2(y, x) * 180 / Math.PI + 360) % 360;
  const labels = ["N", "NE", "E", "SE", "S", "SW", "W", "NW"];
  return labels[Math.round(bearing / 45) % labels.length];
}

function bearingDegrees(from: [number, number], to: [number, number]) {
  const y = Math.sin((to[1] - from[1]) * Math.PI / 180) * Math.cos(to[0] * Math.PI / 180);
  const x =
    Math.cos(from[0] * Math.PI / 180) * Math.sin(to[0] * Math.PI / 180)
    - Math.sin(from[0] * Math.PI / 180)
      * Math.cos(to[0] * Math.PI / 180)
      * Math.cos((to[1] - from[1]) * Math.PI / 180);
  return (Math.atan2(y, x) * 180 / Math.PI + 360) % 360;
}

function haversineMeters(first: [number, number], second: [number, number]) {
  const radius = 6371000;
  const lat1 = first[0] * Math.PI / 180;
  const lat2 = second[0] * Math.PI / 180;
  const deltaLat = (second[0] - first[0]) * Math.PI / 180;
  const deltaLng = (second[1] - first[1]) * Math.PI / 180;
  const value =
    Math.sin(deltaLat / 2) ** 2
    + Math.cos(lat1) * Math.cos(lat2) * Math.sin(deltaLng / 2) ** 2;
  return radius * 2 * Math.atan2(Math.sqrt(value), Math.sqrt(1 - value));
}

function routeDistanceMeters(points: [number, number][], startIndex: number, endIndex: number) {
  let distance = 0;
  const boundedStart = Math.max(0, Math.min(startIndex, points.length - 1));
  const boundedEnd = Math.max(boundedStart, Math.min(endIndex, points.length - 1));
  for (let index = boundedStart; index < boundedEnd; index += 1) {
    distance += haversineMeters(points[index], points[index + 1]);
  }
  return distance;
}

function formatNavigationDistance(meters: number) {
  if (meters >= 950) {
    return `${(meters / 1000).toFixed(1)} km`;
  }
  return `${Math.max(20, Math.round(meters / 10) * 10)} m`;
}

function formatManeuver(maneuver: string) {
  if (maneuver === "left") {
    return "Turn left";
  }
  if (maneuver === "right") {
    return "Turn right";
  }
  if (maneuver === "sharp_left") {
    return "Make a sharp left";
  }
  if (maneuver === "sharp_right") {
    return "Make a sharp right";
  }
  if (maneuver === "depart") {
    return "Start driving";
  }
  return "Continue";
}

function turnFromBearings(previous: number, next: number) {
  const delta = ((next - previous + 540) % 360) - 180;
  if (delta <= -125) {
    return "sharp_left";
  }
  if (delta < -35) {
    return "left";
  }
  if (delta >= 125) {
    return "sharp_right";
  }
  if (delta > 35) {
    return "right";
  }
  return "continue";
}

function nearestRouteIndex(points: [number, number][], latitude: number, longitude: number) {
  return points.reduce((bestIndex, point, index) => {
    const bestPoint = points[bestIndex] ?? point;
    const distance = haversineMeters(point, [latitude, longitude]);
    const bestDistance = haversineMeters(bestPoint, [latitude, longitude]);
    return distance < bestDistance ? index : bestIndex;
  }, 0);
}

function MapNavigationController({
  ambulance,
  route,
  driverMode,
  focusKey,
}: {
  ambulance?: AmbulanceSummary | null;
  route?: MissionRoutePayload | null;
  driverMode: boolean;
  focusKey: string;
}) {
  const map = useMap();

  useEffect(() => {
    if (!route?.polyline?.length) {
      return;
    }
    const routeBounds = latLngBounds(route.polyline);
    map.fitBounds(routeBounds, {
      animate: true,
      duration: 0.8,
      maxZoom: driverMode ? 17 : 15,
      padding: driverMode ? [70, 70] : [40, 40],
    });
    if (driverMode && ambulance) {
      window.setTimeout(() => {
        map.flyTo([ambulance.latitude, ambulance.longitude], 17, {
          animate: true,
          duration: 0.9,
        });
      }, 350);
    }
  }, [ambulance?.id, ambulance?.latitude, ambulance?.longitude, driverMode, focusKey, map, route?.polyline]);

  return null;
}

  async function loadIcuBeds(hospitalId: string): Promise<IcuBed[]> {
    try {
      const response = await apiFetch(`/api/admin/hospitals/${hospitalId}/icu-beds`);
      if (!response.ok) {
        throw new Error("ICU bed request failed");
      }
      const data: IcuBed[] = await response.json();
      setIcuBeds(data);
      setIcuCensusBeds((current) => {
        const patientBeds = data.filter((bed) => bed.patient);
        if (activeUserId !== "super-admin") {
          return patientBeds;
        }
        return [...current.filter((bed) => bed.hospital_id !== hospitalId), ...patientBeds]
          .sort((first, second) => first.hospital_id.localeCompare(second.hospital_id) || first.bed_no.localeCompare(second.bed_no));
      });
      setSelectedBedId((current) => current && data.some((bed) => bed.id === current) ? current : data[0]?.id ?? null);
      setBackendOnline(true);
      return data;
    } catch (err) {
      if (err instanceof TypeError) {
        setBackendOnline(false);
        setError(BACKEND_OFFLINE_MESSAGE);
      } else {
        setError(err instanceof Error ? err.message : "Unexpected ICU bed error");
      }
      return [];
    }
  }

  async function saveSelectedBed() {
    const selectedBed = icuBeds.find((bed) => bed.id === selectedBedId);
    if (!selectedBed) {
      return;
    }
    const hasPatientDraft = [
      bedPatientNo,
      bedPatientName,
      bedPatientIdentifier,
      bedPatientDob,
      bedPatientAge,
      bedPatientSex,
      bedPatientBloodType,
      bedPatientCondition,
      bedPatientDiagnosis,
      bedPatientVitals,
      bedPatientMedications,
      bedPatientAllergies,
      bedPatientInfectionRisk,
      bedPatientContact,
      bedPatientNextOfKin,
      bedPatientAddress,
      bedPatientNotes,
    ].some((value) => value.trim().length > 0) || bedPatientIsolation;
    if (hasPatientDraft && !bedPatientName.trim()) {
      showNotice("Patient Name Required", "Enter a patient name before assigning this bed.", "Update Details", "warning");
      return;
    }
    const parsedAge = bedPatientAge.trim() ? Number(bedPatientAge) : undefined;
    if (parsedAge !== undefined && (!Number.isFinite(parsedAge) || parsedAge < 0 || parsedAge > 130)) {
      showNotice("Valid Age Required", "Enter an age between 0 and 130 before saving this patient record.", "Update Details", "warning");
      return;
    }
    const payload = hasPatientDraft
      ? {
          status: bedStatus === "available" ? "occupied" : bedStatus,
          patient_no: bedPatientNo.trim() || `PAT-${selectedBed.bed_no}`,
          patient_name: bedPatientName.trim(),
          patient_identifier_value: bedPatientIdentifier.trim() || undefined,
          patient_date_of_birth: bedPatientDob.trim() || undefined,
          patient_age: parsedAge,
          patient_sex: bedPatientSex.trim() || undefined,
          patient_blood_type: bedPatientBloodType.trim() || undefined,
          patient_condition: bedPatientCondition.trim() || "under observation",
          diagnosis: bedPatientDiagnosis.trim() || undefined,
          vitals: parseVitalsText(bedPatientVitals),
          medications: parseMedicationText(bedPatientMedications),
          allergies: bedPatientAllergies.trim() || undefined,
          infection_risk: bedPatientInfectionRisk.trim() || undefined,
          isolation_required: bedPatientIsolation,
          emergency_contact: bedPatientContact.trim() || undefined,
          next_of_kin: bedPatientNextOfKin.trim() || undefined,
          address: bedPatientAddress.trim() || undefined,
          notes: bedPatientNotes.trim() || undefined,
        }
      : { status: bedStatus };
    const response = await apiFetch(`/api/admin/icu-beds/${selectedBed.id}`, {
      method: "PATCH",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify(payload),
    });
    if (!response.ok) {
      setError("ICU bed update failed");
      return;
    }
    setBedEditorDirty(false);
    await loadIcuBeds(selectedBed.hospital_id);
    await loadDashboard(true);
  }

  function validateTransferPatientPacket() {
    const missing: string[] = [];
    if (!transferPatientName.trim()) {
      missing.push("patient name");
    }
    if (!transferPatientAge.trim()) {
      missing.push("age");
    }
    const numericAge = Number(transferPatientAge);
    if (transferPatientAge.trim() && (!Number.isFinite(numericAge) || numericAge < 0 || numericAge > 130)) {
      missing.push("valid age");
    }
    if (!transferPatientBloodType.trim()) {
      missing.push("blood type");
    }
    if (!transferPatientContact.trim()) {
      missing.push("emergency contact");
    }
    if (missing.length) {
      showNotice(
        "Patient Details Required",
        `Add ${missing.join(", ")} before sending this transfer request. The receiving hospital needs this packet to reserve the correct ICU bed.`,
        "Update Details",
        "warning",
      );
      return false;
    }
    return true;
  }

      confirmLabel: "Send Request",
      cancelLabel: "Cancel",
      onConfirm: async () => {

        await sendTransferRequest(recommendation);

      onCancel: () => setBlockingModal(null),

  }

  async function sendTransferRequest(recommendation: Recommendation) {
    if (!validateTransferPatientPacket()) {
      return;
    }
    const response = await apiFetch("/api/admin/transfers", {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        origin_hospital_id: originId,
        destination_hospital_id: recommendation.destination_hospital_id,
        required_icu_type: requiredIcu,
        patient_condition: {
          condition_type: conditionType,
          oxygen_saturation_band: oxygen,
          blood_pressure_band: bloodPressure,
          consciousness_level: consciousness,
          ventilator_required: ventilatorRequired,
        },
        patient_name: transferPatientName.trim(),
        patient_identifier_value: transferPatientIdentifier.trim() || undefined,
        patient_date_of_birth: transferPatientDob.trim() || undefined,
        patient_age: Number(transferPatientAge),
        patient_sex: transferPatientSex === "unknown" ? undefined : transferPatientSex,
        patient_blood_type: transferPatientBloodType.trim(),
        patient_allergies: transferPatientAllergies.trim() || undefined,
        patient_emergency_contact: transferPatientContact.trim(),
        patient_diagnosis: transferPatientDiagnosis.trim() || conditionType,
        patient_vitals: parseVitalsText(transferPatientVitals),
        patient_medications: parseMedicationText(transferPatientMedications),
        patient_infection_risk: transferPatientInfectionRisk === "none" ? undefined : transferPatientInfectionRisk,
        patient_isolation_required: transferPatientIsolation,
        handover_notes: transferPatientAllergies.trim() || undefined,
        notes: `Created from recommendation rank ${recommendation.rank}`,
      }),
    });
    if (!response.ok) {
      setError("Transfer request creation failed");
      return;
    }
    const transfer: TransferSummary = await response.json();
    await loadDashboard();
    showNotice(
      "Request Sent",
      `Transfer request sent to ${recommendation.destination_name}. Request ID: ${transfer.id.slice(0, 8)}.`,
      "Continue",
      "success",
    );

  async function transferAction(transferId: string, action: "accept" | "reject" | "assign-ambulance") {

  async function executeTransferAction(transferId: string, action: "accept" | "reject" | "assign-ambulance") {
    const response = await apiFetch(`/api/admin/transfers/${transferId}/${action}`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
      },
      body: action === "assign-ambulance" ? undefined : JSON.stringify({ notes: action }),
    });
    if (!response.ok) {
      setError(`${action} failed`);
      return;
    }
    const updatedTransfer: TransferSummary = await response.json();
    await loadDashboard();
    const bedHospitalId = updatedTransfer.destination_hospital_id || icuHospitalId;
    if (bedHospitalId) {
      const freshBeds = await loadIcuBeds(bedHospitalId);
      if (updatedTransfer.assigned_bed_id && freshBeds.some((bed) => bed.id === updatedTransfer.assigned_bed_id)) {
        setSelectedBedId(updatedTransfer.assigned_bed_id);
        setMapCommandView("beds");
      }
    }
    if (action === "accept") {
      showNotice(
        "Request Accepted",
        updatedTransfer.ambulance_id
          ? `Transfer accepted. Bed ${updatedTransfer.assigned_bed_id ?? "reserved"} and ambulance ${updatedTransfer.ambulance_id} assigned.`
          : `Transfer accepted. Bed ${updatedTransfer.assigned_bed_id ?? "reserved"} assigned; waiting for ambulance.`,
        "Continue",
        "success",
      );
    }
  }

  function parseEventDetails(detailsJson: string | null) {
    if (!detailsJson) {
      return [];

    try {

  function parseMissionRoute(transfer?: TransferSummary | null): MissionRoutePayload | null {
    if (!transfer?.route_payload_json) {
      return null;
    }
    try {
      const parsed = JSON.parse(transfer.route_payload_json) as {
        pickup_route?: MissionRoutePayload;
        destination_route?: MissionRoutePayload;
        mission_route?: MissionRoutePayload;
      };
      if (["ambulance_assigned", "ambulance_en_route_to_pickup"].includes(transfer.status)) {
        return parsed.pickup_route ?? parsed.mission_route ?? null;
      }
      return parsed.destination_route ?? parsed.mission_route ?? null;
    } catch {
      return null;
    }
  }

  function parseDispatchPayload(transfer?: TransferSummary | null): DispatchPayload | null {
    if (!transfer?.route_payload_json) {
      return null;
    }
    try {
      const parsed = JSON.parse(transfer.route_payload_json) as { dispatch?: DispatchPayload };
      return parsed.dispatch ?? null;
    } catch {
      return null;
    }
  }

  function googleMapsUrl(transfer?: TransferSummary | null) {
    if (!transfer) {
      return null;
    }
    const pickup = `${transfer.pickup_latitude},${transfer.pickup_longitude}`;
    const dropoff = `${transfer.dropoff_latitude},${transfer.dropoff_longitude}`;
    if (
      transfer.pickup_latitude == null ||
      transfer.pickup_longitude == null ||
      transfer.dropoff_latitude == null ||
      transfer.dropoff_longitude == null
    ) {
      return null;
    }
    return `https://www.google.com/maps/dir/?api=1&origin=${pickup}&destination=${dropoff}&travelmode=driving`;
  }

  function formatDateTime(value?: string) {
    if (!value) {
      return "";
    }
    return new Intl.DateTimeFormat(undefined, {
      hour: "2-digit",
      minute: "2-digit",
      month: "short",
      day: "numeric",
    }).format(new Date(value));
  }

  function parseVitalsText(value: string) {
    const vitals: Record<string, string> = {};
    value
      .split(/[,\n]/)
      .map((part) => part.trim())
      .filter(Boolean)
      .forEach((part) => {
        const [rawKey, ...rawValue] = part.split(/[:=]/);
        const key = rawKey?.trim();
        const nextValue = rawValue.join(":").trim();
        if (key && nextValue) {
          vitals[key] = nextValue;
        }
      });
    return Object.keys(vitals).length ? vitals : undefined;
  }

  function parseMedicationText(value: string) {
    return value
      .split(/[,\n]/)
      .map((item) => item.trim())
      .filter(Boolean);
  }

  function formatVitals(value: Record<string, unknown> | null | undefined) {
    if (!value) {
      return "";
    }
    return Object.entries(value).map(([key, item]) => `${key}: ${String(item)}`).join(", ");
  }

  function processDashboardNotifications(data: DashboardSummary) {
    const seenTransfers = seenTransfersByUserRef.current[activeUserId] ?? new Set<string>();
    const statusMap = transferStatusByUserRef.current[activeUserId] ?? new Map<string, string>();
    const ambulanceMap = transferAmbulanceByUserRef.current[activeUserId] ?? new Map<string, string | null>();
    const user = users.find((item) => item.id === activeUserId);
    const hospitalId = user?.hospital_id ?? null;
    if (!user && activeUserId !== "super-admin") {
      return;
    }

    if (blockingModal) {
      return;
    }

    for (const transfer of data.transfers) {
      const previousStatus = statusMap.get(transfer.id);
      const previousAmbulance = ambulanceMap.get(transfer.id);
      const isIncoming = Boolean(hospitalId && transfer.destination_hospital_id === hospitalId);
      const isOutgoing = Boolean(hospitalId && transfer.origin_hospital_id === hospitalId);
      const isRelevant = activeUserId === "super-admin" || isIncoming || isOutgoing;

      if (isIncoming && !seenTransfers.has(transfer.id) && transfer.status === "pending_destination_acceptance") {
        seenTransfers.add(transfer.id);
        showNotice(
          "Request Arrived",
          `${hospitalName(transfer.origin_hospital_id)} sent ${transfer.patient_name ?? "a patient"} for ${transfer.required_icu_type}.`,
          "Review Request",
          "warning",
          () => {
            setConsoleTab("requests");
            setHighlightedTransferId(transfer.id);
          },
        );
        break;
      }

      if (
        isOutgoing &&
        previousStatus === "pending_destination_acceptance" &&
        ["accepted_pending_ambulance", "ambulance_assigned"].includes(transfer.status)
      ) {
        showNotice(
          "Request Accepted",
          `${hospitalName(transfer.destination_hospital_id)} accepted your transfer request.`,
          "Continue",
          "success",
        );
        break;
      }

      if (isRelevant && !previousAmbulance && transfer.ambulance_id) {
        showNotice(
          "Ambulance Assigned",
          `Ambulance ${transfer.ambulance_id} has been assigned to the transfer from ${hospitalName(transfer.origin_hospital_id)} to ${hospitalName(transfer.destination_hospital_id)}.`,
          "Acknowledge",
          "success",
        );
        break;
      }
    }

    data.transfers.forEach((transfer) => {
      seenTransfers.add(transfer.id);
      statusMap.set(transfer.id, transfer.status);
      ambulanceMap.set(transfer.id, transfer.ambulance_id);
    });
    seenTransfersByUserRef.current[activeUserId] = seenTransfers;
    transferStatusByUserRef.current[activeUserId] = statusMap;
    transferAmbulanceByUserRef.current[activeUserId] = ambulanceMap;
  }

  function transferStage(transfer: TransferSummary) {
    const stages = [
      "pending_destination_acceptance",
      "ambulance_assigned",
      "ambulance_en_route_to_pickup",
      "en_route_to_destination",
      "completed",
    ];
    if (transfer.status === "rejected" || transfer.status === "cancelled") {
      return -1;
    }
    const index = stages.indexOf(transfer.status);
    if (index >= 0) {
      return index;
    }
    if (transfer.status === "accepted_pending_ambulance") {
      return 1;
    }
    return 0;
  }

  function transferProgressState(transfer: TransferSummary) {
    if (transfer.status === "completed") {
      return { label: "Completed", className: "completed" };
    }
    if (transfer.status === "rejected" || transfer.status === "cancelled") {
      return { label: formatStatus(transfer.status), className: "stopped" };
    }
    if (transfer.status === "pending_destination_acceptance") {
      return { label: "Pending acceptance", className: "pending" };
    }
    return { label: "In progress", className: "active" };
  }

  function timelineClass(transfer: TransferSummary, index: number) {
    const stage = transferStage(transfer);
    if (transfer.status === "rejected" || transfer.status === "cancelled") {
      return index === 0 ? "stopped" : "";
    }
    if (transfer.status === "completed") {
      return "done";
    }
    if (index < stage) {
      return "done";
    }
    if (index === stage) {
      return "current";
    }
    return "";
  }

                <span>Required ICU</span>
                <strong>{selectedTransferDetail.required_icu_type}</strong>
              </div>
              <div>
                <span>Ambulance</span>
                <strong>{selectedTransferDetail.ambulance_id ?? "Not assigned"}</strong>
              </div>
              <div>
                <span>Assigned bed</span>
                <strong>{selectedTransferDetail.assigned_bed_id ? (icuBeds.find((bed) => bed.id === selectedTransferDetail.assigned_bed_id)?.bed_no ?? selectedTransferDetail.assigned_bed_id) : "Not assigned"}</strong>
              </div>
            </div>

            {selectedDispatch && (
              <>
                <h3>Dispatch Decision</h3>
                <div className="detail-grid">
                  <div>
                    <span>Model</span>
                    <strong>{selectedDispatch.model ?? "dispatch model"}</strong>
                  </div>
                  <div>
                    <span>Score</span>
                    <strong>{selectedDispatch.score?.toFixed?.(2) ?? "-"}</strong>
                  </div>
                  <div>
                    <span>Pickup ETA</span>
                    <strong>{selectedDispatch.estimated_pickup_minutes ?? "-"} min</strong>
                  </div>
                  <div>
                    <span>Pickup risk</span>
                    <strong>{selectedDispatch.pickup_risk_score?.toFixed?.(2) ?? "-"}</strong>
                  </div>
                  <div>
                    <span>Coverage impact</span>
                    <strong>{selectedDispatch.coverage_penalty?.toFixed?.(2) ?? "-"}</strong>
                  </div>
                </div>
                {selectedDispatch.explanation?.length ? (
                  <div className="dispatch-reasons">
                    {selectedDispatch.explanation.map((reason) => (
                      <span key={reason}>{reason}</span>
                    ))}
                  </div>
                ) : null}
              </>
            )}

            <h3>Patient Packet</h3>
            <div className="detail-grid">
              <div>
                <span>Name</span>
                <strong>{selectedTransferDetail.patient_name ?? "Unnamed patient"}</strong>
              </div>
              <div>
                <span>Identifier</span>
                <strong>{selectedTransferDetail.patient_identifier_value ?? "-"}</strong>
              </div>
              <div>
                <span>Age / sex</span>
                <strong>{selectedTransferDetail.patient_age ?? "-"} / {selectedTransferDetail.patient_sex ?? "-"}</strong>
              </div>
              <div>
                <span>Date of birth</span>
                <strong>{selectedTransferDetail.patient_date_of_birth ?? "-"}</strong>
              </div>
              <div>
                <span>Blood type</span>
                <strong>{selectedTransferDetail.patient_blood_type ?? "-"}</strong>
              </div>
              <div>
                <span>Condition</span>
                <strong>{selectedTransferDetail.patient_condition}</strong>
              </div>
              <div>
                <span>Diagnosis</span>
                <strong>{selectedTransferDetail.patient_diagnosis ?? "-"}</strong>
              </div>
              <div>
                <span>Vitals</span>
                <strong>{formatVitals(selectedTransferDetail.patient_vitals) || "-"}</strong>
              </div>
              <div>
                <span>Medications</span>
                <strong>{selectedTransferDetail.patient_medications?.length ? selectedTransferDetail.patient_medications.join(", ") : "-"}</strong>
              </div>
              <div>
                <span>Allergies / alerts</span>
                <strong>{selectedTransferDetail.patient_allergies ?? "None recorded"}</strong>
              </div>
              <div>
                <span>Infection / isolation</span>
                <strong>{selectedTransferDetail.patient_infection_risk ?? "none"} / {selectedTransferDetail.patient_isolation_required ? "isolation required" : "no isolation"}</strong>
              </div>
              <div>
                <span>Emergency contact</span>
                <strong>{selectedTransferDetail.patient_emergency_contact ?? "-"}</strong>
              </div>
            </div>

            <h3>Event Timeline</h3>
            {eventsLoading ? (
              <div className="muted">Loading timeline...</div>
            ) : transferEvents.length ? (
              <div className="event-list">
                {transferEvents.map((event) => {
                  const details = parseEventDetails(event.details_json);
                  return (
                    <div className="event-row" key={event.id}>
                      <div className="event-dot" />
                      <div>
                        <strong>{formatStatus(event.event_type)}</strong>
                        <span>{event.message}</span>
                        <small>
                          {formatDateTime(event.created_at)}
                          {event.actor_user_id ? ` / ${event.actor_user_id}` : ""}
                          {` / ${formatStatus(event.status)}`}
                        </small>
                        {details.length > 0 && (
                          <ul>
                            {details.slice(0, 6).map((detail) => (
                              <li key={detail}>{detail}</li>
                            ))}
                          </ul>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            ) : (
              <div className="muted">No timeline events recorded</div>
            )}
          </section>
        </div>
      )}
    </main>
  );
}

export default App;
